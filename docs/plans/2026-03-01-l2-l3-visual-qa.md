## Task 7 — Regression Guard: Playwright Smoke Tests

**Files:**
- Create: `e2e/l2-l3-smoke.spec.ts`

**Step 1: Check if Playwright is installed**
```bash
ls package.json | xargs grep playwright
```
If missing: `npm install -D @playwright/test && npx playwright install chromium`

**Step 2: Write smoke test**
```ts
// e2e/l2-l3-smoke.spec.ts
import { test, expect } from '@playwright/test';

const BASE = process.env.PLAYWRIGHT_BASE_URL || 'https://hq.saimor.world';

test.describe('L2 DepartmentLayer smoke', () => {
  test('renders center orb and at least one moon', async ({ page }) => {
    await page.goto(BASE);
    // Wait for L2 to be visible (department layer is default view)
    await page.waitForSelector('[data-testid^="space-"]', { timeout: 10000 });
    // Center orb exists
    await expect(page.locator('text=Department Core')).toBeVisible();
    // At least one moon rendered
    const moons = page.locator('[data-testid^="space-"]');
    await expect(moons).toHaveCountGreaterThan(0);
  });
});

test.describe('L3 SpaceLayer smoke', () => {
  test('clicking moon navigates to SpaceLayer and shows folder orbs', async ({ page }) => {
    await page.goto(BASE);
    await page.waitForSelector('[data-testid^="space-"]', { timeout: 10000 });
    // Click the first moon
    const firstMoon = page.locator('[data-testid^="space-"]').first();
    await firstMoon.click();
    // Wait for L3
    await page.waitForSelector('[data-testid^="folder-"]', { timeout: 10000 });
    await expect(page.locator('text=Folder Cluster')).toBeVisible();
    const folders = page.locator('[data-testid^="folder-"]');
    await expect(folders).toHaveCountGreaterThan(0);
  });
});
```

**Step 3: Add Playwright config if missing**
```ts
// playwright.config.ts
import { defineConfig } from '@playwright/test';
export default defineConfig({
  testDir: './e2e',
  use: { headless: true, viewport: { width: 1280, height: 800 } },
});
```

**Step 4: Run tests**
```bash
npx playwright test e2e/l2-l3-smoke.spec.ts
```
Expected: 2 tests pass (or skip if backend unavailable — add `test.skip(!!process.env.CI, 'needs live server')`)

**Step 5: Commit**
```bash
git add e2e/ playwright.config.ts package.json package-lock.json
git commit -m "test(e2e): add L2/L3 Playwright smoke tests for demo regression guard"
```
